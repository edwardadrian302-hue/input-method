cmake_minimum_required(VERSION 3.22)
project(chaoxi_rime LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
get_filename_component(ROOT "${CMAKE_CURRENT_LIST_DIR}/../.." ABSOLUTE)
set(DEPS "${ROOT}/vendor/android-deps")
set(RIME "${ROOT}/vendor/librime")
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(BUILD_TESTING OFF CACHE BOOL "" FORCE)
set(YAML_CPP_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(YAML_CPP_BUILD_TOOLS OFF CACHE BOOL "" FORCE)
set(LEVELDB_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(LEVELDB_BUILD_BENCHMARKS OFF CACHE BOOL "" FORCE)
set(LEVELDB_INSTALL OFF CACHE BOOL "" FORCE)
set(ENABLE_TOOLS OFF CACHE BOOL "" FORCE)
set(OPENCC_ENABLE_INSTALL OFF CACHE BOOL "" FORCE)
set(OPENCC_DICT_FORMAT text CACHE STRING "" FORCE)
set(ENABLE_GTEST OFF CACHE BOOL "" FORCE)
set(BUILD_DOCUMENTATION OFF CACHE BOOL "" FORCE)
add_subdirectory("${DEPS}/yaml-cpp" yaml-cpp EXCLUDE_FROM_ALL)
add_subdirectory("${DEPS}/leveldb" leveldb EXCLUDE_FROM_ALL)
add_subdirectory("${DEPS}/opencc" opencc EXCLUDE_FROM_ALL)
file(MAKE_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/include/opencc" "${CMAKE_CURRENT_BINARY_DIR}/include/rime")
file(GLOB OPENCC_HEADERS "${DEPS}/opencc/src/*.h" "${DEPS}/opencc/src/*.hpp")
file(COPY ${OPENCC_HEADERS} DESTINATION "${CMAKE_CURRENT_BINARY_DIR}/include/opencc")
set(RIME_DATA_DIR "rime-data")
configure_file("${RIME}/src/rime/build_config.h.in" "${CMAKE_CURRENT_BINARY_DIR}/include/rime/build_config.h")
file(GLOB RIME_SOURCES CONFIGURE_DEPENDS
    "${RIME}/src/*.cc" "${RIME}/src/rime/*.cc" "${RIME}/src/rime/algo/*.cc"
    "${RIME}/src/rime/config/*.cc" "${RIME}/src/rime/dict/*.cc"
    "${RIME}/src/rime/gear/*.cc" "${RIME}/src/rime/lever/*.cc")
add_library(rime SHARED ${RIME_SOURCES})
target_include_directories(rime PRIVATE "${RIME}/src" "${RIME}/include"
    "${CMAKE_CURRENT_BINARY_DIR}/include" "${DEPS}/boost_1_89_0"
    "${DEPS}/opencc/deps/marisa-0.3.1/include")
target_compile_definitions(rime PRIVATE RIME_VERSION="1.17.0" RIME_EXTRA_MODULES= RIME_EXPORTS YAML_CPP_STATIC_DEFINE Opencc_BUILT_AS_STATIC BOOST_REGEX_NO_LIB)
target_link_libraries(rime PRIVATE yaml-cpp leveldb libopencc marisa dl)
target_link_options(rime PRIVATE "-Wl,-z,max-page-size=16384")
install(TARGETS rime LIBRARY DESTINATION lib)
