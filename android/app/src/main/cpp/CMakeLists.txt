cmake_minimum_required(VERSION 3.22.1)
project(chaoxi_android LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
get_filename_component(ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../../../../.." ABSOLUTE)
set(RIME_PREFIX "${ROOT}/.tools/android-rime/${ANDROID_ABI}")
if(NOT EXISTS "${RIME_PREFIX}/lib/librime.so")
    message(FATAL_ERROR "Android Rime is missing. Run scripts/prepare-android-rime.ps1 first.")
endif()
add_library(rime SHARED IMPORTED)
set_target_properties(rime PROPERTIES IMPORTED_LOCATION "${RIME_PREFIX}/lib/librime.so")
add_library(chaoxi SHARED bridge.cpp "${ROOT}/src/engine.cpp")
target_include_directories(chaoxi PRIVATE "${ROOT}/src" "${ROOT}/vendor/librime/src")
target_link_libraries(chaoxi rime)
target_link_options(chaoxi PRIVATE "-Wl,-z,max-page-size=16384")
